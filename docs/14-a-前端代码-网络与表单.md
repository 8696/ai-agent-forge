## 14.1 Axios 实例 + 拦截器（鉴权 + 错误处理）

```typescript
// web/lib/api.ts
import axios, { AxiosError, type InternalAxiosRequestConfig } from 'axios';
import { message } from 'antd';

export const api = axios.create({
  baseURL: '/api',
  timeout: 30_000,
});

// 请求拦截器：自动带 JWT
api.interceptors.request.use((config: InternalAxiosRequestConfig) => {
  const token = localStorage.getItem('jwt');
  if (token) {
    config.headers.set('Authorization', `Bearer ${token}`);
  }
  return config;
});

// 响应拦截器：401 跳登录 + 错误统一提示
api.interceptors.response.use(
  (response) => response,
  (error: AxiosError) => {
    const status = error.response?.status;
    const data = error.response?.data as { code?: string; message?: string } | undefined;

    if (status === 401) {
      localStorage.removeItem('jwt');
      window.location.href = '/admin/login';
      return Promise.reject(error);
    }

    if (status === 403) {
      message.error('没有权限');
    } else if (status >= 500) {
      message.error(data?.message ?? '服务异常，请稍后重试');
    } else if (data?.message) {
      message.error(data.message);
    }

    return Promise.reject(error);
  },
);

// 类型化的 API 调用
export async function getTraces(params: { tenant_id?: string; limit?: number }) {
  const { data } = await api.get<{ traces: Trace[] }>('/admin/traces', { params });
  return data.traces;
}

export async function uploadDocument(file: File, onProgress?: (p: number) => void) {
  const formData = new FormData();
  formData.append('file', file);
  const { data } = await api.post<{ id: string }>('/v1/knowledge/upload', formData, {
    headers: { 'Content-Type': 'multipart/form-data' },
    onUploadProgress: (e) => {
      if (e.total) onProgress?.(Math.round((e.loaded * 100) / e.total));
    },
  });
  return data;
}
```

## 14.2 AntD Form + Zod 校验

```typescript
// web/components/forms/DocumentUploadForm.tsx
import { Form, Input, Upload, Button, message } from 'antd';
import { useForm } from 'antd/es/form/Form';
import { z } from 'zod';
import { uploadDocument } from '@/lib/api';

const formSchema = z.object({
  title: z.string().min(1, '标题不能为空').max(500),
  category: z.enum(['policy', 'faq', 'manual', 'other']),
  description: z.string().max(2000).optional()
});

type FormValues = z.infer<typeof formSchema>;

export function DocumentUploadForm({ onSuccess }: { onSuccess: () => void }) {
  const [form] = useForm<FormValues>();
  const [submitting, setSubmitting] = useState(false);

  async function onFinish(values: FormValues) {
    // 二次校验（防止 AntD 漏校验）
    const parsed = formSchema.safeParse(values);
    if (!parsed.success) {
      message.error('表单数据不合法');
      return;
    }

    const file = (values as any).file?.[0]?.originFileObj as File | undefined;
    if (!file) {
      message.error('请选择文件');
      return;
    }

    setSubmitting(true);
    try {
      await uploadDocument(file);
      message.success('上传成功，正在异步索引');
      form.resetFields();
      onSuccess();
    } catch (e) {
      // 拦截器已统一报错
    } finally {
      setSubmitting(false);
    }
  }

  return (
    <Form form={form} onFinish={onFinish} layout="vertical">
      <Form.Item
        name="title"
        label="文档标题"
        rules={[{ required: true, message: '标题不能为空' }, { max: 500 }]}
      >
        <Input placeholder="例如：退款政策 v3" />
      </Form.Item>

      <Form.Item
        name="category"
        label="分类"
        rules={[{ required: true }]}
      >
        <Input placeholder="policy / faq / manual / other" />
      </Form.Item>

      <Form.Item name="description" label="描述">
        <Input.TextArea rows={3} maxLength={2000} showCount />
      </Form.Item>

      <Form.Item>
        <Upload maxCount={1} beforeUpload={() => false} accept=".pdf,.docx,.md,.txt">
          <Button>选择文件（PDF / Word / Markdown / TXT）</Button>
        </Upload>
      </Form.Item>

      <Form.Item>
        <Button type="primary" htmlType="submit" loading={submitting}>
          上传
        </Button>
      </Form.Item>
    </Form>
  );
}
```
