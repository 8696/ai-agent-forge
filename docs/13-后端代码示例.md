## 13.1 Sequelize 模型定义 + 调用

### 模型定义（装饰器风格）

```typescript
// src/models/Tenant.ts
import {
  Table,
  Column,
  Model,
  PrimaryKey,
  Default,
  DataType
} from 'sequelize-typescript';

@Table({ tableName: 'tenants', timestamps: true })
export class Tenant extends Model {
  @PrimaryKey
  @Default(DataType.UUIDV4)
  @Column(DataType.UUID)
  declare id: string;

  @Column({ type: DataType.STRING(200), allowNull: false })
  name!: string;

  @Default('free')
  @Column(DataType.ENUM('free', 'pro', 'enterprise'))
  plan!: 'free' | 'pro' | 'enterprise';

  @Column(DataType.JSON)
  settings?: Record<string, unknown>;

  @Column(DataType.STRING(255))
  api_key_hash?: string;
}
```

### 启动时自动跑迁移

```typescript
// src/server.ts
import { Umzug, SequelizeStorage } from 'umzug';
import path from 'path';
import { sequelize } from '@/db';

export async function runMigrations() {
  const umzug = new Umzug({
    migrations: {
      glob: [path.resolve(__dirname, 'migrations/*.js'), { cwd: __dirname }],
      resolve: ({ name, path: migrationPath, context }) => {
        // eslint-disable-next-line @typescript-eslint/no-require-imports
        const migration = require(migrationPath as string);
        return {
          name,
          up: async () => migration.up(context.sequelize.getQueryInterface(), Sequelize),
          down: async () => migration.down(context.sequelize.getQueryInterface(), Sequelize)
        };
      }
    },
    context: { sequelize },
    storage: new SequelizeStorage({ sequelize }),
    logger: console
  });

  const pending = await umzug.pending();
  if (pending.length > 0) {
    console.log(`Pending migrations: ${pending.map(p => p.name).join(', ')}`);
    await umzug.up();
    console.log('Migrations applied');
  }
}

// 在 server.ts 启动流程里
await runMigrations();
await sequelize.authenticate();  // 测一下连接
```

## 13.5 Koa 路由 + Vercel AI SDK（串起来）

```typescript
// src/gateway/chat-routes.ts
import Router from '@koa/router';
import { z } from 'zod';
import { streamText, convertToCoreMessages, type UIMessage } from 'ai';
import { anthropic } from '@ai-sdk/anthropic';
import { Readable } from 'node:stream';

const chatInputSchema = z.object({
  tenant_id: z.string().min(1),
  session_id: z.string().uuid(),
  messages: z.array(z.object({
    role: z.enum(['user', 'assistant']),
    content: z.string()
  }))
});

export function buildChatRouter() {
  const router = new Router();

  // 同步版（给非浏览器调用方用）
  router.post('/api/v1/chat', async (ctx) => {
    const { tenant_id, session_id, messages } = chatInputSchema.parse(ctx.request.body);

    const result = await streamText({
      model: anthropic('claude-sonnet-4-5'),
      system: await loadSystemPrompt(tenant_id),
      messages: convertToCoreMessages(messages as UIMessage[])
    });

    const text = await result.text;  // 完整文本（流式版本用 toDataStreamResponse）
    ctx.body = { reply: text, session_id };
  });

  // 流式版（SSE，给浏览器用）
  router.post('/api/v1/chat/stream', async (ctx) => {
    const { tenant_id, messages } = chatInputSchema.parse(ctx.request.body);

    const result = streamText({
      model: anthropic('claude-sonnet-4-5'),
      system: await loadSystemPrompt(tenant_id),
      messages: convertToCoreMessages(messages as UIMessage[])
    });

    const response = result.toDataStreamResponse();
    ctx.status = response.status;
    response.headers.forEach((v, k) => ctx.set(k, v));
    ctx.body = response.body ? Readable.fromWeb(response.body) : null;
  });

  return router;
}

// src/server.ts
import Koa from 'koa';
import bodyParser from 'koa-bodyparser';
import cors from '@koa/cors';
import { runMigrations } from '@/db';
import { buildChatRouter } from '@/gateway/chat-routes';

async function bootstrap() {
  // 1. 跑迁移
  await runMigrations();

  // 2. 启 Koa（Vercel AI SDK 的 provider 在路由内按需拿）
  const app = new Koa();
  app.use(cors());
  app.use(bodyParser());
  app.use(buildChatRouter().routes());

  app.listen(3000);
  console.log('Server running on http://localhost:3000');
}

bootstrap().catch(console.error);

## 13.6 SSE 流式聊天（Koa + Vercel AI SDK）

> 聊天界面要「边生成边显示」，用 SSE。后端用 Vercel AI SDK 的 `streamText()` 一句拿流式，前端用 `@ai-sdk/react` 的 `useChat()` 一行接。
> 比直接调 Anthropic SDK 简单 ~60 行。

```typescript
// src/gateway/chat-stream-routes.ts
import Router from '@koa/router';
import { z } from 'zod';
import { streamText, convertToCoreMessages, type UIMessage } from 'ai';
import { anthropic } from '@ai-sdk/anthropic';
import { Readable } from 'node:stream';

const chatInputSchema = z.object({
  tenant_id: z.string().min(1),     // 租户号
  session_id: z.string().uuid(),
  messages: z.array(z.object({
    role: z.enum(['user', 'assistant']),
    content: z.string()
  }))                                // 来自前端 useChat 的 messages 数组
});

export function buildChatStreamRouter() {
  const router = new Router();

  router.post('/api/v1/chat/stream', async (ctx) => {
    const { tenant_id, session_id, messages } = chatInputSchema.parse(ctx.request.body);

    // Vercel AI SDK 一句调用
    const result = streamText({
      model: anthropic('claude-sonnet-4-5'),
      system: await loadSystemPrompt(tenant_id),
      messages: convertToCoreMessages(messages as UIMessage[]),
      // 可选：工具调用
      // tools: { order_query: tool({...}) },
      // maxTokens: 1024,
      // temperature: 0.2,
    });

    // Vercel AI SDK 返回标准 Response 对象，转成 Koa 流
    const response = result.toDataStreamResponse();
    ctx.status = response.status;
    response.headers.forEach((v, k) => ctx.set(k, v));
    ctx.body = response.body ? Readable.fromWeb(response.body) : null;
  });

  return router;
}
```

> **对比之前**（手写 Anthropic SDK + SSE 响应头 + event 循环）：从 30+ 行降到 5 行。
> **之前手写的循环**（for await + event 分类 + 手动写 `event: text`）**全部不需要**——Vercel AI SDK 自动生成正确的 SSE 格式。

## 13.7 文档索引（同步处理 · 当前 MVP）

> 当前项目规模下，文档上传后**同步**处理（API 等几秒直接返回）。不引入 BullMQ + Redis 队列，省运维负担。
> 后面真有大文件要异步，再引入队列。

```typescript
// src/gateway/knowledge-routes.ts（同步处理版本）
import Router from '@koa/router';
import { koaBody } from 'koa-body';
import path from 'path';
import { promises as fs } from 'fs';
import { Document } from '@/models/Document';
import { config } from '@/config';
import { nanoid } from 'nanoid';
import { ingestDocument } from '@/rag/ingest';

export function buildKnowledgeRouter() {
  const router = new Router();

  router.post('/api/admin/knowledge/upload', koaBody({
    multipart: true,
    formidable: {
      uploadDir: path.resolve(config.DATA_DIR, 'tmp'),
      keepExtensions: true,
      maxFileSize: 50 * 1024 * 1024  // 50 MB
    }
  }), async (ctx) => {
    const file = ctx.request.files?.['file'];
    if (!file) throw new ValidationError('Missing file');

    const tenantId = ctx.state.user.tenant_id;

    // 1. 永久保存
    const ext = path.extname(file.originalFilename ?? '');
    const storedName = `${nanoid()}${ext}`;
    const destPath = path.resolve(config.DATA_DIR, 'uploads', tenantId, storedName);
    await fs.mkdir(path.dirname(destPath), { recursive: true });
    await fs.rename(file.filepath, destPath);

    // 2. 写 documents 表，状态 indexing
    const doc = await Document.create({
      tenant_id: tenantId,
      title: file.originalFilename ?? 'untitled',
      source: storedName,
      file_type: detectFileType(ext),
      file_path: destPath,
      status: 'indexing',
      uploaded_by: ctx.state.user.sub
    });

    try {
      // 3. 同步处理：分块 + Embedding + 入向量库
      await ingestDocument({ id: doc.id, tenantId, filePath: destPath });

      // 4. 改状态为 ready
      await Document.update(
        { status: 'ready' },
        { where: { id: doc.id } }
      );

      ctx.body = { id: doc.id, status: 'ready' };
    } catch (err) {
      // 5. 失败改状态
      await Document.update(
        { status: 'failed' },
        { where: { id: doc.id } }
      );
      throw err;
    }
  });

  return router;
}

function detectFileType(ext: string): 'pdf' | 'word' | 'markdown' | 'txt' {
  if (ext === '.pdf') return 'pdf';
  if (['.doc', '.docx'].includes(ext)) return 'word';
  if (['.md', '.markdown'].includes(ext)) return 'markdown';
  return 'txt';
}
```

> **未来扩展**（如果文档很大 / 并发上传很多导致用户等太久）：把 `ingestDocument()` 调用包成 BullMQ 任务即可，业务逻辑不用改。那时候再把 BullMQ + Redis 加到依赖里，按这一节原来的生产者 + 消费者模式写。

## 13.8 用户注册 + 鉴权（bcrypt + JWT）

### 注册 + 登录

```typescript
// src/gateway/auth-routes.ts
import Router from '@koa/router';
import bcrypt from 'bcrypt';
import jwt from 'jsonwebtoken';
import { z } from 'zod';
import { User } from '@/models/User';
import { config } from '@/config';
import { ValidationError, AuthError } from '@/lib/errors';

const registerSchema = z.object({
  email: z.string().email(),
  password: z.string().min(8, '密码至少 8 位'),
  tenant_id: z.string().uuid(),
  role: z.enum(['admin', 'agent', 'customer']).default('agent')
});

export function buildAuthRouter() {
  const router = new Router();

  router.post('/api/v1/auth/register', async (ctx) => {
    const input = registerSchema.parse(ctx.request.body);

    const exists = await User.findOne({ where: { email: input.email } });
    if (exists) throw new ValidationError('邮箱已注册');

    const password_hash = await bcrypt.hash(input.password, 12);

    const user = await User.create({
      email: input.email,
      password_hash,
      tenant_id: input.tenant_id,
      role: input.role
    });

    ctx.body = { id: user.id, email: user.email, role: user.role };
  });

  // 登录：必须带 tenant_code（登录的租户隔离）+ login_type（区分后台 / 前台）
  router.post('/api/v1/auth/login', async (ctx) => {
    const { email, password, tenant_code, login_type } = z.object({
      email: z.string().email(),
      password: z.string(),
      tenant_code: z.string().min(3).max(32),
      login_type: z.enum(['admin', 'agent', 'customer'])  // 'admin' 和 'agent' 走后台；'customer' 走前台
    }).parse(ctx.request.body);

    // 1. 先用 tenant_code 查租户
    const tenant = await Tenant.findOne({ where: { code: tenant_code } });
    if (!tenant) throw new AuthError('邮箱或密码错误');  // 不泄露租户是否存在

    // 2. 在该租户内查用户
    const user = await User.findOne({
      where: { email, tenant_id: tenant.id }
    });
    if (!user) throw new AuthError('邮箱或密码错误');

    // 3. 验证角色与 login_type 一致
    if (user.role !== login_type) {
      // 'admin' 和 'agent' 都允许后台登录
      if (!(login_type === 'admin' && (user.role === 'admin' || user.role === 'agent'))) {
        throw new AuthError('邮箱或密码错误');
      }
    }

    // 4. 验证密码
    const ok = await bcrypt.compare(password, user.password_hash!);
    if (!ok) throw new AuthError('邮箱或密码错误');

    // 5. 签发 JWT（payload 含 tenant_id）
    const token = jwt.sign(
      { sub: user.id, tenant_id: user.tenant_id, role: user.role },
      config.JWT_SECRET,
      { expiresIn: config.JWT_EXPIRES_IN }
    );

    ctx.body = {
      token,
      user: { id: user.id, email: user.email, role: user.role, tenant_code: tenant.code }
    };
  });

  // SaaS 平台方建租户（自动初始化第一个 admin）
  router.post('/api/v1/system/tenants', jwtSystemAuth, async (ctx) => {
    const { code, name, admin_email, admin_password } = z.object({
      code: z.string().regex(/^[a-z0-9-]{3,32}$/, '租户号必须小写字母数字短横线'),
      name: z.string().min(1).max(200),
      admin_email: z.string().email(),
      admin_password: z.string().min(8)
    }).parse(ctx.request.body);

    // 1. 检查租户号唯一
    const exists = await Tenant.findOne({ where: { code } });
    if (exists) throw new ValidationError('租户号已被使用');

    // 2. 在事务里建租户 + admin
    const result = await sequelize.transaction(async (tx) => {
      const tenant = await Tenant.create({ code, name }, { transaction: tx });
      const admin_password_hash = await bcrypt.hash(admin_password, 12);
      const admin = await User.create({
        email: admin_email,
        password_hash: admin_password_hash,
        tenant_id: tenant.id,
        role: 'admin'
      }, { transaction: tx });
      return { tenant, admin };
    });

    ctx.body = {
      tenant: { id: result.tenant.id, code: result.tenant.code, name: result.tenant.name },
      admin: { id: result.admin.id, email: result.admin.email, role: 'admin' }
    };
  });

  return router;
}

// src/gateway/jwt-middleware.ts
import jwt from 'jsonwebtoken';
import { config } from '@/config';
import { AuthError } from '@/lib/errors';

export async function jwtAuth(ctx: any, next: any) {
  const auth = ctx.headers.authorization;
  if (!auth?.startsWith('Bearer ')) {
    throw new AuthError('Missing Bearer token');
  }

  const token = auth.slice(7);
  try {
    const payload = jwt.verify(token, config.JWT_SECRET) as {
      sub: string;
      tenant_id: string;
      role: string;
    };
    ctx.state.user = payload;
  } catch (e) {
    throw new AuthError('Invalid token');
  }

  await next();
}

## 13.9 文件上传（@koa/multer 处理 PDF / Word）

```typescript
// src/gateway/knowledge-routes.ts
import Router from '@koa/router';
import { koaBody } from 'koa-body';
import path from 'path';
import { promises as fs } from 'fs';
import { enqueueIndexDocument } from '@/queue/document-index-queue';
import { Document } from '@/models/Document';
import { config } from '@/config';
import { nanoid } from 'nanoid';

export function buildKnowledgeRouter() {
  const router = new Router();

  router.post('/api/v1/knowledge/upload', koaBody({
    multipart: true,
    formidable: {
      uploadDir: path.resolve(config.DATA_DIR, 'tmp'),
      keepExtensions: true,
      maxFileSize: 50 * 1024 * 1024  // 50 MB
    }
  }), async (ctx) => {
    const file = ctx.request.files?.['file'];
    if (!file) throw new ValidationError('Missing file');

    const tenantId = ctx.state.user.tenant_id;

    // 1. 永久保存到 data/uploads/
    const ext = path.extname(file.originalFilename ?? '');
    const storedName = `${nanoid()}${ext}`;
    const destPath = path.resolve(config.DATA_DIR, 'uploads', tenantId, storedName);
    await fs.mkdir(path.dirname(destPath), { recursive: true });
    await fs.rename(file.filepath, destPath);

    // 2. 写 documents 表（状态 uploaded）
    const doc = await Document.create({
      tenant_id: tenantId,
      title: file.originalFilename ?? 'untitled',
      source: storedName,
      file_type: detectFileType(ext),
      file_path: destPath,
      status: 'uploaded',
      uploaded_by: ctx.state.user.sub
    });

    // 3. 入队异步跑 embedding
    await enqueueIndexDocument({
      documentId: doc.id,
      tenantId,
      filePath: destPath
    });

    ctx.body = { id: doc.id, status: 'uploaded' };
  });

  return router;
}

function detectFileType(ext: string): 'pdf' | 'word' | 'markdown' | 'txt' {
  if (ext === '.pdf') return 'pdf';
  if (['.doc', '.docx'].includes(ext)) return 'word';
  if (['.md', '.markdown'].includes(ext)) return 'markdown';
  return 'txt';
}

## 13.10 MCP Server 端示例（怎么写一个 MCP 服务让 Agent 来连）

```typescript
// mcp-servers/order-system/src/server.ts
import { Server } from '@modelcontextprotocol/sdk/server/index.js';
import { StdioServerTransport } from '@modelcontextprotocol/sdk/server/stdio.js';
import {
  CallToolRequestSchema,
  ListToolsRequestSchema
} from '@modelcontextprotocol/sdk/types.js';

const server = new Server(
  { name: 'order-system', version: '1.0.0' },
  { capabilities: { tools: {} } }
);

// 列出所有工具
server.setRequestHandler(ListToolsRequestSchema, async () => ({
  tools: [
    {
      name: 'order_query',
      description: '查询订单',
      inputSchema: {
        type: 'object',
        properties: {
          order_id: { type: 'string', description: '订单 ID' },
          user_id: { type: 'string', description: '用户 ID' }
        }
      }
    },
    {
      name: 'order_update_address',
      description: '修改收货地址（敏感操作）',
      inputSchema: {
        type: 'object',
        properties: {
          order_id: { type: 'string' },
          new_address: { type: 'string' }
        },
        required: ['order_id', 'new_address']
      }
    }
  ]
}));

server.setRequestHandler(CallToolRequestSchema, async (request) => {
  const { name, arguments: args } = request.params;

  if (name === 'order_query') {
    const result = await queryOrderDatabase(args);
    return { content: [{ type: 'text', text: JSON.stringify(result) }] };
  }

  if (name === 'order_update_address') {
    const result = await markAddressChangeForReview(args);
    return { content: [{ type: 'text', text: JSON.stringify(result) }] };
  }

  throw new Error(`Unknown tool: ${name}`);
});

async function queryOrderDatabase(args: any) {
  return { order_id: args.order_id, status: 'shipped' };
}

async function markAddressChangeForReview(args: any) {
  return { status: 'pending_review', message: '请等待人工确认' };
}

// stdio 模式，主项目通过 child_process 启动这个进程
const transport = new StdioServerTransport();
await server.connect(transport);
console.error('Order system MCP server started on stdio');
