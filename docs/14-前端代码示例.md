## 14.1 Axios 实例 + 拦截器（鉴权 + 错误处理）

```typescript
// web/lib/api.ts
import axios, { AxiosError, type InternalAxiosRequestConfig } from 'axios';
import { message } from 'antd';

export const api = axios.create({
  baseURL: '/api',
  timeout: 30_000
});

// 请求拦截器：自动带 JWT
api.interceptors.request.use((config: InternalAxiosRequestConfig) => {
  const token = localStorage.getItem('jwt');
  if (token) {
    config.headers.set('Authorization', `Bearer ${token}`);
  }
  return config;
});

// 响应拦截器：401 跳登录 + 错误统一提示
api.interceptors.response.use(
  (response) => response,
  (error: AxiosError) => {
    const status = error.response?.status;
    const data = error.response?.data as { code?: string; message?: string } | undefined;

    if (status === 401) {
      localStorage.removeItem('jwt');
      window.location.href = '/admin/login';
      return Promise.reject(error);
    }

    if (status === 403) {
      message.error('没有权限');
    } else if (status >= 500) {
      message.error(data?.message ?? '服务异常，请稍后重试');
    } else if (data?.message) {
      message.error(data.message);
    }

    return Promise.reject(error);
  }
);

// 类型化的 API 调用
export async function getTraces(params: { tenant_id?: string; limit?: number }) {
  const { data } = await api.get<{ traces: Trace[] }>('/admin/traces', { params });
  return data.traces;
}

export async function uploadDocument(file: File, onProgress?: (p: number) => void) {
  const formData = new FormData();
  formData.append('file', file);
  const { data } = await api.post<{ id: string }>('/v1/knowledge/upload', formData, {
    headers: { 'Content-Type': 'multipart/form-data' },
    onUploadProgress: (e) => {
      if (e.total) onProgress?.(Math.round((e.loaded * 100) / e.total));
    }
  });
  return data;
}
```

## 14.2 AntD Form + Zod 校验

```typescript
// web/components/forms/DocumentUploadForm.tsx
import { Form, Input, Upload, Button, message } from 'antd';
import { useForm } from 'antd/es/form/Form';
import { z } from 'zod';
import { uploadDocument } from '@/lib/api';

const formSchema = z.object({
  title: z.string().min(1, '标题不能为空').max(500),
  category: z.enum(['policy', 'faq', 'manual', 'other']),
  description: z.string().max(2000).optional()
});

type FormValues = z.infer<typeof formSchema>;

export function DocumentUploadForm({ onSuccess }: { onSuccess: () => void }) {
  const [form] = useForm<FormValues>();
  const [submitting, setSubmitting] = useState(false);

  async function onFinish(values: FormValues) {
    // 二次校验（防止 AntD 漏校验）
    const parsed = formSchema.safeParse(values);
    if (!parsed.success) {
      message.error('表单数据不合法');
      return;
    }

    const file = (values as any).file?.[0]?.originFileObj as File | undefined;
    if (!file) {
      message.error('请选择文件');
      return;
    }

    setSubmitting(true);
    try {
      await uploadDocument(file);
      message.success('上传成功，正在异步索引');
      form.resetFields();
      onSuccess();
    } catch (e) {
      // 拦截器已统一报错
    } finally {
      setSubmitting(false);
    }
  }

  return (
    <Form form={form} onFinish={onFinish} layout="vertical">
      <Form.Item
        name="title"
        label="文档标题"
        rules={[{ required: true, message: '标题不能为空' }, { max: 500 }]}
      >
        <Input placeholder="例如：退款政策 v3" />
      </Form.Item>

      <Form.Item
        name="category"
        label="分类"
        rules={[{ required: true }]}
      >
        <Input placeholder="policy / faq / manual / other" />
      </Form.Item>

      <Form.Item name="description" label="描述">
        <Input.TextArea rows={3} maxLength={2000} showCount />
      </Form.Item>

      <Form.Item>
        <Upload maxCount={1} beforeUpload={() => false} accept=".pdf,.docx,.md,.txt">
          <Button>选择文件（PDF / Word / Markdown / TXT）</Button>
        </Upload>
      </Form.Item>

      <Form.Item>
        <Button type="primary" htmlType="submit" loading={submitting}>
          上传
        </Button>
      </Form.Item>
    </Form>
  );
}
```

## 14.3 聊天界面（用 `@ai-sdk/react` 的 `useChat` · 1 行接 SSE）

> **从 40 行手写 SSE parser 简化到 1 个 hook**。Vercel AI SDK 的 React 包自动处理：SSE 解析、状态管理、断线重连、AbortController。

```typescript
// web/pages/widget/ChatPage.tsx
import { useChat } from '@ai-sdk/react';
import { useParams } from 'react-router-dom';
import { useEffect, useState } from 'react';
import { ChatMessage } from '@/components/ChatMessage';

export function ChatPage() {
  const { tenantCode } = useParams<{ tenantCode: string }>();
  const [sessionId] = useState(() => crypto.randomUUID());

  // ⭐ Vercel AI SDK 的 useChat —— 1 行接 SSE
  const { messages, input, handleInputChange, handleSubmit, isLoading, stop, reload, error } = useChat({
    id: sessionId,                       // 同一 session 维持上下文
    api: '/api/v1/chat/stream',           // 后端 SSE 端点
    body: { tenant_id: tenantCode },       // 每次请求带租户号
    initialMessages: [],                  // 首次进入空；可从后端拉历史填充
    onError: (err) => {
      message.error(err.message || '聊天失败');
    }
  });

  return (
    <div className="chat-container">
      {/* Header */}
      <header>
        <span>🏢 公司名 · 客服 · 在线</span>
        <Button icon={<CloseOutlined />} onClick={onClose} />
      </header>

      {/* 消息列表 */}
      <div className="messages">
        {messages.map(m => (
          <ChatMessage key={m.id} message={m} />
        ))}
        {isLoading && (
          <div className="loading">
            <Spin /> <span>思考中...</span>
            <Button size="small" onClick={() => stop()}>停止</Button>
          </div>
        )}
      </div>

      {/* 输入框 */}
      <form onSubmit={handleSubmit}>
        <Input.TextArea
          value={input}
          onChange={handleInputChange}
          placeholder="输入你的问题..."
          autoSize={{ minRows: 1, maxRows: 4 }}
          onPressEnter={e => { if (!e.shiftKey) { e.preventDefault(); handleSubmit(e as any); } }}
        />
        <Button type="primary" htmlType="submit" loading={isLoading}>
          发送
        </Button>
      </form>
    </div>
  );
}
```

> **对比之前**（40 行手写 SSE parser + AbortController + 状态管理）：**现在 1 个 hook**。useChat 内部自动维护 `messages` 数组、自动解析 SSE、自动重连、自动管理 abort。

### useChat 提供的关键 API

| API | 用途 |
|-----|------|
| `messages` | 完整消息数组（含用户 + AI），自动更新 |
| `input` | 输入框当前内容 |
| `handleInputChange` | 绑到 `<input>` 的 onChange |
| `handleSubmit` | 绑到 `<form>` 的 onSubmit |
| `isLoading` | 是否在等流式响应 |
| `stop()` | 中止当前流 |
| `reload()` | 重新发送最后一条 |
| `error` | 错误对象 |
| `append(msg)` | 编程式发消息（不通过表单） |

### 加载历史消息（v1 可选）

```typescript
import { useAuthStore } from '@/lib/store/auth';

export function RequireAuth({ children }: { children: React.ReactNode }) {
  const location = useLocation();
```

## 14.5 Markdown 渲染（含代码高亮 + GFM）

```typescript
// web/components/Markdown.tsx
import ReactMarkdown from 'react-markdown';
import remarkGfm from 'remark-gfm';
import { Prism as SyntaxHighlighter } from 'react-syntax-highlighter';
import { oneDark } from 'react-syntax-highlighter/dist/esm/styles/prism';
import { Button, message } from 'antd';
import { CopyOutlined } from '@ant-design/icons';

export function Markdown({ content }: { content: string }) {
  return (
    <ReactMarkdown
      remarkPlugins={[remarkGfm]}
      components={{
        code({ node, inline, className, children, ...props }: any) {
          const match = /language-(\w+)/.exec(className ?? '');
          const code = String(children).replace(/\n$/, '');

          if (!inline && match) {
            return (
              <div style={{ position: 'relative' }}>
                <Button
                  size="small"
                  icon={<CopyOutlined />}
                  style={{ position: 'absolute', top: 8, right: 8 }}
                  onClick={() => {
                    navigator.clipboard.writeText(code);
                    message.success('已复制');
                  }}
                />
                <SyntaxHighlighter language={match[1]} style={oneDark}>
                  {code}
                </SyntaxHighlighter>
              </div>
            );
          }

          return <code className={className} {...props}>{children}</code>;
        },
        table({ children }) {
          return <table style={{ borderCollapse: 'collapse' }}>{children}</table>;
        },
        th({ children }) {
          return <th style={{ border: '1px solid #d9d9d9', padding: 8 }}>{children}</th>;
        },
        td({ children }) {
          return <td style={{ border: '1px solid #d9d9d9', padding: 8 }}>{children}</td>;
        }
      }}
    >
      {content}
    </ReactMarkdown>
  );
}
```

> **注意**：额外依赖 `react-syntax-highlighter`。

## 14.7 错误边界

```typescript
// web/components/ErrorBoundary.tsx
import { Component, type ReactNode } from 'react';
import { Result, Button } from 'antd';

interface Props { children: ReactNode; fallback?: ReactNode; }
interface State { hasError: boolean; error?: Error; }

export class ErrorBoundary extends Component<Props, State> {
  state: State = { hasError: false };

  static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error };
  }

  componentDidCatch(error: Error, info: React.ErrorInfo) {
    console.error('React error boundary caught:', error, info);
    // TODO: 发送到 trace / observability
  }

  render() {
    if (this.state.hasError) {
      if (this.props.fallback) return this.props.fallback;
      return (
        <Result
          status="error"
          title="页面出错了"
          subTitle={this.state.error?.message}
          extra={
            <Button type="primary" onClick={() => window.location.reload()}>
              刷新页面
            </Button>
          }
        />
      );
    }
    return this.props.children;
  }
}

// 路由根挂上
<ErrorBoundary>
  <ConfigProvider locale={zhCN} theme={theme}>
    <RouterProvider router={router} />
  </ConfigProvider>
</ErrorBoundary>
```

## 14.8 Zustand store（按业务拆）

```typescript
// web/lib/store/auth.ts
import { create } from 'zustand';

interface User {
  id: string;
  email: string;
  role: 'admin' | 'agent' | 'customer';
  tenant_id: string;
}

interface AuthState {
  user: User | null;
  loading: boolean;
  login: (email: string, password: string) => Promise<void>;
  logout: () => void;
  fetchUser: () => Promise<void>;
}

export const useAuthStore = create<AuthState>((set) => ({
  user: null,
  loading: true,

  async login(email, password) {
    const { data } = await api.post<{ token: string; user: User }>('/v1/auth/login', {
      email,
      password
    });
    localStorage.setItem('jwt', data.token);
    set({ user: data.user });
  },

  logout() {
    localStorage.removeItem('jwt');
    set({ user: null });
  },

  async fetchUser() {
    const token = localStorage.getItem('jwt');
    if (!token) {
      set({ user: null, loading: false });
      return;
    }
    try {
      const { data } = await api.get<{ user: User }>('/v1/auth/me');
      set({ user: data.user, loading: false });
    } catch {
      set({ user: null, loading: false });
    }
  }
}));
```

```typescript
interface ChatState {
  streamingText: string;
  streamingToolCall: { name: string } | null;

  setToolCall: (tool: { name: string } | null) => void;

  streamingToolCall: null,
```
