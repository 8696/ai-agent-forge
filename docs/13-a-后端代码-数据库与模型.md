## 13.1 Sequelize 模型定义 + 调用

### 模型定义（装饰器风格）

```typescript
// src/models/Tenant.ts
import { Table, Column, Model, PrimaryKey, Default, DataType } from 'sequelize-typescript';

@Table({ tableName: 'tenants', timestamps: true })
export class Tenant extends Model {
  @PrimaryKey
  @Default(DataType.UUIDV4)
  @Column(DataType.UUID)
  declare id: string;

  @Column({ type: DataType.STRING(200), allowNull: false })
  name!: string;

  @Default('free')
  @Column(DataType.ENUM('free', 'pro', 'enterprise'))
  plan!: 'free' | 'pro' | 'enterprise';

  @Column(DataType.JSON)
  settings?: Record<string, unknown>;

  @Column(DataType.STRING(255))
  api_key_hash?: string;
}
```

JSON 字段读写示例（**Sequelize 比 Prisma 自然**）：

```typescript
// 读 JSON 字段里的某个子键
const settings = await Tenant.findByPk(id);
const lang = settings?.settings?.['language']; // 直接当对象访问

// 写 JSON 字段
await settings?.update({
  settings: { ...settings.settings, language: 'zh-CN' },
});
```

### 调用模型（仓储层）

```typescript
// src/repositories/tenant-repository.ts
import { Op } from 'sequelize';
import { Tenant } from '@/models/Tenant';

export async function findTenantById(id: string) {
  return await Tenant.findByPk(id);
}

export async function listTenantsByPlan(plan: 'free' | 'pro' | 'enterprise', limit = 50) {
  return await Tenant.findAll({
    where: { plan },
    limit,
    order: [['created_at', 'DESC']],
  });
}

export async function createTenant(data: { name: string; plan?: 'free' | 'pro' | 'enterprise' }) {
  return await Tenant.create(data);
}

// 多租户通用过滤（项目里所有 search 都要带 tenant_id）
export async function findSessionsByTenant(tenantId: string, since?: Date) {
  return await Session.findAll({
    where: {
      tenant_id: tenantId,
      ...(since ? { started_at: { [Op.gte]: since } } : {}),
    },
  });
}
```

### 复杂查询回退到原生 SQL

```typescript
// src/repositories/document-repository.ts
import { sequelize } from '@/db';

export async function searchDocumentsByMetadata(
  tenantId: string,
  metadataKey: string,
  metadataValue: string,
) {
  return await sequelize.query(
    `SELECT id, title, metadata
         FROM documents
         WHERE tenant_id = :tenantId
           AND JSON_EXTRACT(metadata, :path) = :value
           AND status = 'ready'
         LIMIT 100`,
    {
      replacements: {
        tenantId,
        path: `$.${metadataKey}`,
        value: metadataValue,
      },
      type: 'SELECT',
    },
  );
}
```

## 13.3 数据库初始化 + 迁移运行

### 初始化（启动时跑）

```typescript
// src/db.ts
import { Sequelize } from 'sequelize-typescript';
import * as models from '@/models';

export const sequelize = new Sequelize({
  dialect: 'mysql',
  host: process.env.MYSQL_HOST ?? 'localhost',
  port: Number(process.env.MYSQL_PORT ?? 3306),
  username: process.env.MYSQL_USER ?? 'root',
  password: process.env.MYSQL_PASSWORD ?? '',
  database: process.env.MYSQL_DB ?? 'agent_platform',
  models: Object.values(models),
  logging: process.env.SQL_LOG === '1' ? console.log : false,
});
```

### 启动时自动跑迁移

```typescript
// src/server.ts
import { Umzug, SequelizeStorage } from 'umzug';
import path from 'path';
import { sequelize } from '@/db';

export async function runMigrations() {
  const umzug = new Umzug({
    migrations: {
      glob: [path.resolve(__dirname, 'migrations/*.js'), { cwd: __dirname }],
      resolve: ({ name, path: migrationPath, context }) => {
        // eslint-disable-next-line @typescript-eslint/no-require-imports
        const migration = require(migrationPath as string);
        return {
          name,
          up: async () => migration.up(context.sequelize.getQueryInterface(), Sequelize),
          down: async () => migration.down(context.sequelize.getQueryInterface(), Sequelize),
        };
      },
    },
    context: { sequelize },
    storage: new SequelizeStorage({ sequelize }),
    logger: console,
  });

  const pending = await umzug.pending();
  if (pending.length > 0) {
    console.log(`Pending migrations: ${pending.map((p) => p.name).join(', ')}`);
    await umzug.up();
    console.log('Migrations applied');
  }
}

// 在 server.ts 启动流程里
await runMigrations();
await sequelize.authenticate(); // 测一下连接
```

### 手动生成新迁移

```bash
# 生成迁移模板（按 Sequelize CLI）
npx sequelize-cli migration:generate --name add-tenant-billing-email

# 跑迁移
npx sequelize-cli db:migrate

# 回滚一步
npx sequelize-cli db:migrate:undo

# 回到初始状态
npx sequelize-cli db:migrate:undo:all
```

## 13.4 字段修改迁移（4 种最常见的迁移写法）

### 新增字段

```typescript
// src/migrations/20260101-add-tenant-billing-email.ts
import { DataType } from 'sequelize-typescript';

export async function up(queryInterface: any) {
  await queryInterface.addColumn('tenants', 'billing_email', {
    type: DataType.STRING(255),
    allowNull: true,
  });
}

export async function down(queryInterface: any) {
  await queryInterface.removeColumn('tenants', 'billing_email');
}
```

### 改字段类型

```typescript
// src/migrations/20260201-change-messages-content.ts
import { DataType } from 'sequelize-typescript';

export async function up(queryInterface: any) {
  // MEDIUMTEXT → LONGTEXT（消息变长，需要扩字段）
  await queryInterface.changeColumn('messages', 'content', {
    type: DataType.TEXT('long'),
  });
}

export async function down(queryInterface: any) {
  await queryInterface.changeColumn('messages', 'content', {
    type: DataType.TEXT('medium'),
  });
}
```

### 删字段

```typescript
// src/migrations/20260301-drop-documents-source.ts
import { DataType } from 'sequelize-typescript';

export async function up(queryInterface: any) {
  await queryInterface.removeColumn('documents', 'source');
}

export async function down(queryInterface: any) {
  await queryInterface.addColumn('documents', 'source', {
    type: DataType.STRING(500),
    allowNull: true,
  });
}
```

### 加索引（性能优化）

```typescript
// src/migrations/20260401-add-messages-tenant-index.ts
export async function up(queryInterface: any) {
  await queryInterface.addIndex('messages', ['tenant_id', 'created_at'], {
    name: 'idx_messages_tenant_created',
  });
}

export async function down(queryInterface: any) {
  await queryInterface.removeIndex('messages', 'idx_messages_tenant_created');
}
```

### 删索引

```typescript
export async function up(queryInterface: any) {
  await queryInterface.removeIndex('messages', 'idx_messages_tenant_created');
}

export async function down(queryInterface: any) {
  await queryInterface.addIndex('messages', ['tenant_id', 'created_at'], {
    name: 'idx_messages_tenant_created',
  });
}
```

### 重命名表

```typescript
// src/migrations/20260501-rename-sessions-to-conversations.ts
export async function up(queryInterface: any) {
  await queryInterface.renameTable('sessions', 'conversations');
}

export async function down(queryInterface: any) {
  await queryInterface.renameTable('conversations', 'sessions');
}
```
