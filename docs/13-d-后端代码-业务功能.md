## 13.7 文档索引（同步处理 · 当前 MVP）

> 当前项目规模下，文档上传后**同步**处理（API 等几秒直接返回）。不引入 BullMQ + Redis 队列，省运维负担。
> 后面真有大文件要异步，再引入队列。

```typescript
// src/gateway/knowledge-routes.ts（同步处理版本）
import Router from '@koa/router';
import { koaBody } from 'koa-body';
import path from 'path';
import { promises as fs } from 'fs';
import { Document } from '@/models/Document';
import { config } from '@/config';
import { nanoid } from 'nanoid';
import { ingestDocument } from '@/rag/ingest';

export function buildKnowledgeRouter() {
  const router = new Router();

  router.post(
    '/api/admin/knowledge/upload',
    koaBody({
      multipart: true,
      formidable: {
        uploadDir: path.resolve(config.DATA_DIR, 'tmp'),
        keepExtensions: true,
        maxFileSize: 50 * 1024 * 1024, // 50 MB
      },
    }),
    async (ctx) => {
      const file = ctx.request.files?.['file'];
      if (!file) throw new ValidationError('Missing file');

      const tenantId = ctx.state.user.tenant_id;

      // 1. 永久保存
      const ext = path.extname(file.originalFilename ?? '');
      const storedName = `${nanoid()}${ext}`;
      const destPath = path.resolve(config.DATA_DIR, 'uploads', tenantId, storedName);
      await fs.mkdir(path.dirname(destPath), { recursive: true });
      await fs.rename(file.filepath, destPath);

      // 2. 写 documents 表，状态 indexing
      const doc = await Document.create({
        tenant_id: tenantId,
        title: file.originalFilename ?? 'untitled',
        source: storedName,
        file_type: detectFileType(ext),
        file_path: destPath,
        status: 'indexing',
        uploaded_by: ctx.state.user.sub,
      });

      try {
        // 3. 同步处理：分块 + Embedding + 入向量库
        await ingestDocument({ id: doc.id, tenantId, filePath: destPath });

        // 4. 改状态为 ready
        await Document.update({ status: 'ready' }, { where: { id: doc.id } });

        ctx.body = { id: doc.id, status: 'ready' };
      } catch (err) {
        // 5. 失败改状态
        await Document.update({ status: 'failed' }, { where: { id: doc.id } });
        throw err;
      }
    },
  );

  return router;
}

function detectFileType(ext: string): 'pdf' | 'word' | 'markdown' | 'txt' {
  if (ext === '.pdf') return 'pdf';
  if (['.doc', '.docx'].includes(ext)) return 'word';
  if (['.md', '.markdown'].includes(ext)) return 'markdown';
  return 'txt';
}
```

> **未来扩展**（如果文档很大 / 并发上传很多导致用户等太久）：把 `ingestDocument()` 调用包成 BullMQ 任务即可，业务逻辑不用改。那时候再把 BullMQ + Redis 加到依赖里，按这一节原来的生产者 + 消费者模式写。

## 13.8 用户注册 + 鉴权（bcrypt + JWT）

### 注册 + 登录

```typescript
// src/gateway/auth-routes.ts
import Router from '@koa/router';
import bcrypt from 'bcrypt';
import jwt from 'jsonwebtoken';
import { z } from 'zod';
import { User } from '@/models/User';
import { config } from '@/config';
import { ValidationError, AuthError } from '@/lib/errors';

const registerSchema = z.object({
  email: z.string().email(),
  password: z.string().min(8, '密码至少 8 位'),
  tenant_id: z.string().uuid(),
  role: z.enum(['admin', 'agent', 'customer']).default('agent'),
});

export function buildAuthRouter() {
  const router = new Router();

  router.post('/api/v1/auth/register', async (ctx) => {
    const input = registerSchema.parse(ctx.request.body);

    const exists = await User.findOne({ where: { email: input.email } });
    if (exists) throw new ValidationError('邮箱已注册');

    const password_hash = await bcrypt.hash(input.password, 12);

    const user = await User.create({
      email: input.email,
      password_hash,
      tenant_id: input.tenant_id,
      role: input.role,
    });

    ctx.body = { id: user.id, email: user.email, role: user.role };
  });

  // 登录：必须带 tenant_code（登录的租户隔离）+ login_type（区分后台 / 前台）
  router.post('/api/v1/auth/login', async (ctx) => {
    const { email, password, tenant_code, login_type } = z
      .object({
        email: z.string().email(),
        password: z.string(),
        tenant_code: z.string().min(3).max(32),
        login_type: z.enum(['admin', 'agent', 'customer']), // 'admin' 和 'agent' 走后台；'customer' 走前台
      })
      .parse(ctx.request.body);

    // 1. 先用 tenant_code 查租户
    const tenant = await Tenant.findOne({ where: { code: tenant_code } });
    if (!tenant) throw new AuthError('邮箱或密码错误'); // 不泄露租户是否存在

    // 2. 在该租户内查用户
    const user = await User.findOne({
      where: { email, tenant_id: tenant.id },
    });
    if (!user) throw new AuthError('邮箱或密码错误');

    // 3. 验证角色与 login_type 一致
    if (user.role !== login_type) {
      // 'admin' 和 'agent' 都允许后台登录
      if (!(login_type === 'admin' && (user.role === 'admin' || user.role === 'agent'))) {
        throw new AuthError('邮箱或密码错误');
      }
    }

    // 4. 验证密码
    const ok = await bcrypt.compare(password, user.password_hash!);
    if (!ok) throw new AuthError('邮箱或密码错误');

    // 5. 签发 JWT（payload 含 tenant_id）
    const token = jwt.sign(
      { sub: user.id, tenant_id: user.tenant_id, role: user.role },
      config.JWT_SECRET,
      { expiresIn: config.JWT_EXPIRES_IN },
    );

    ctx.body = {
      token,
      user: { id: user.id, email: user.email, role: user.role, tenant_code: tenant.code },
    };
  });

  // SaaS 平台方建租户（自动初始化第一个 admin）
  router.post('/api/v1/system/tenants', jwtSystemAuth, async (ctx) => {
    const { code, name, admin_email, admin_password } = z
      .object({
        code: z.string().regex(/^[a-z0-9-]{3,32}$/, '租户号必须小写字母数字短横线'),
        name: z.string().min(1).max(200),
        admin_email: z.string().email(),
        admin_password: z.string().min(8),
      })
      .parse(ctx.request.body);

    // 1. 检查租户号唯一
    const exists = await Tenant.findOne({ where: { code } });
    if (exists) throw new ValidationError('租户号已被使用');

    // 2. 在事务里建租户 + admin
    const result = await sequelize.transaction(async (tx) => {
      const tenant = await Tenant.create({ code, name }, { transaction: tx });
      const admin_password_hash = await bcrypt.hash(admin_password, 12);
      const admin = await User.create(
        {
          email: admin_email,
          password_hash: admin_password_hash,
          tenant_id: tenant.id,
          role: 'admin',
        },
        { transaction: tx },
      );
      return { tenant, admin };
    });

    ctx.body = {
      tenant: { id: result.tenant.id, code: result.tenant.code, name: result.tenant.name },
      admin: { id: result.admin.id, email: result.admin.email, role: result.admin.role },
    };
  });

  return router;
}
```

### 鉴权中间件

```typescript
// src/gateway/jwt-middleware.ts
import jwt from 'jsonwebtoken';
import { config } from '@/config';
import { AuthError } from '@/lib/errors';

export async function jwtAuth(ctx: any, next: any) {
  const auth = ctx.headers.authorization;
  if (!auth?.startsWith('Bearer ')) {
    throw new AuthError('Missing Bearer token');
  }

  const token = auth.slice(7);
  try {
    const payload = jwt.verify(token, config.JWT_SECRET) as {
      sub: string;
      tenant_id: string;
      role: string;
    };
    ctx.state.user = payload;
  } catch (e) {
    throw new AuthError('Invalid token');
  }

  await next();
}

// 路由里挂鉴权
router.get('/api/admin/dashboard', jwtAuth, async (ctx) => {
  // ctx.state.user 已经填好
});
```

> **关于本项目实际做法**：本项目用 **3 个独立 `requireXxxUser` 中间件**（`requireSystemUser` / `requireAdmin` / `requireCustomer`），按子系统拆分，**禁止共享**。详见 `03-b-技术栈-后端栈.md` §B.5.7 和 `16-a-需求-多租户与登录.md` A4 节的端点清单。

## 13.9 文件上传（@koa/multer 处理 PDF / Word）

```typescript
// src/gateway/knowledge-routes.ts
import Router from '@koa/router';
import { koaBody } from 'koa-body';
import path from 'path';
import { promises as fs } from 'fs';
import { enqueueIndexDocument } from '@/queue/document-index-queue';
import { Document } from '@/models/Document';
import { config } from '@/config';
import { nanoid } from 'nanoid';

export function buildKnowledgeRouter() {
  const router = new Router();

  router.post(
    '/api/v1/knowledge/upload',
    koaBody({
      multipart: true,
      formidable: {
        uploadDir: path.resolve(config.DATA_DIR, 'tmp'),
        keepExtensions: true,
        maxFileSize: 50 * 1024 * 1024, // 50 MB
      },
    }),
    async (ctx) => {
      const file = ctx.request.files?.['file'];
      if (!file) throw new ValidationError('Missing file');

      const tenantId = ctx.state.user.tenant_id;

      // 1. 永久保存到 data/uploads/
      const ext = path.extname(file.originalFilename ?? '');
      const storedName = `${nanoid()}${ext}`;
      const destPath = path.resolve(config.DATA_DIR, 'uploads', tenantId, storedName);
      await fs.mkdir(path.dirname(destPath), { recursive: true });
      await fs.rename(file.filepath, destPath);

      // 2. 写 documents 表（状态 uploaded）

        tenant_id: tenantId,

        documentId: doc.id,

    },
  );

  return router;
}

  if (ext === '.pdf') return 'pdf';

  if (['.md', '.markdown'].includes(ext)) return 'markdown';
  return 'txt';
}
```
