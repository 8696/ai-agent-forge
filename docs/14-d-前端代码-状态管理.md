## 14.8 Zustand store（按业务拆）

```typescript
// web/lib/store/auth.ts
import { create } from 'zustand';

interface User {
  id: string;
  email: string;
  role: 'admin' | 'agent' | 'customer';
  tenant_id: string;
}

interface AuthState {
  user: User | null;
  loading: boolean;
  login: (email: string, password: string) => Promise<void>;
  logout: () => void;
  fetchUser: () => Promise<void>;
}

export const useAuthStore = create<AuthState>((set) => ({
  user: null,
  loading: true,

  async login(email, password) {
    const { data } = await api.post<{ token: string; user: User }>('/v1/auth/login', {
      email,
      password,
    });
    localStorage.setItem('jwt', data.token);
    set({ user: data.user });
  },

  logout() {
    localStorage.removeItem('jwt');
    set({ user: null });
  },

  async fetchUser() {
    const token = localStorage.getItem('jwt');
    if (!token) {
      set({ user: null, loading: false });
      return;
    }
    try {
      const { data } = await api.get<{ user: User }>('/v1/auth/me');
      set({ user: data.user, loading: false });
    } catch {
      set({ user: null, loading: false });
    }
  },
}));
```

聊天 store（每个 session 一份）：

```typescript
// web/lib/store/chat.ts
import { create } from 'zustand';

interface ChatState {
  streamingText: string;
  streamingToolCall: { name: string } | null;
  appendText: (delta: string) => void;
  setToolCall: (tool: { name: string } | null) => void;
  reset: () => void;
}

export const useChatStore = create<ChatState>((set) => ({
  streamingText: '',
  streamingToolCall: null,
  appendText: (delta) => set((s) => ({ streamingText: s.streamingText + delta })),
  setToolCall: (tool) => set({ streamingToolCall: tool }),
  reset: () => set({ streamingText: '', streamingToolCall: null }),
}));
```

## 14.9 AntD 主题（中文 locale + 自定义 token）

```typescript
// web/App.tsx
import { ConfigProvider, theme as antdTheme } from 'antd';
import zhCN from 'antd/locale/zh_CN';
import 'dayjs/locale/zh-cn';

const theme = {
  token: {
    colorPrimary: '#1677ff',
    colorSuccess: '#52c41a',
    colorWarning: '#faad14',
    colorError: '#ff4d4f',
    borderRadius: 6,
    fontSize: 14
  },
  components: {
    Button: { borderRadius: 6 },
    Card: { borderRadiusLG: 8 }
  }
};

export function App() {
  return (
    <ConfigProvider
      locale={zhCN}
      theme={{
        ...antdTheme,
        token: theme.token,
        components: theme.components
      }}
    >
      {/* ... */}
    </ConfigProvider>
  );
}
```

> 用 dayjs 的 `zh-cn` locale 让所有日期组件显示中文（AntD 5 默认 dayjs，依赖在前面 `03-a-技术栈-工程化规范.md` §A.8 已列）。
