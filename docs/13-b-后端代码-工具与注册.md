## 13.2 工具定义（Builtin + MCP）

### 内置工具（项目自己的工具）

```typescript
// src/tools/builtin/order-query.ts
import { z } from 'zod';
import type { Tool } from '@/tools/types';

// 用 Zod 定义入参 + 类型自动推导
export const orderQueryInputSchema = z.object({
  order_id: z.string().optional(),
  user_id: z.string().optional(),
  since: z.string().datetime().optional(),
});

export type OrderQueryInput = z.infer<typeof orderQueryInputSchema>;

export const orderQueryTool: Tool = {
  name: 'order_query',
  description: '查询用户的订单（按订单 ID 或按时间范围）',
  inputSchema: orderQueryInputSchema,
  execute: async (rawInput: unknown) => {
    const input = orderQueryInputSchema.parse(rawInput); // 运行时校验
    return await callOrderService(input);
  },
};
```

### MCP 工具（外部系统）

```typescript
// src/tools/mcp/connect-order-system.ts
import { Client } from '@modelcontextprotocol/sdk/client/index.js';
import { StdioClientTransport } from '@modelcontextprotocol/sdk/client/stdio.js';

export async function connectOrderSystem() {
  const transport = new StdioClientTransport({
    command: 'node',
    args: ['./mcp-servers/order-server/index.js'],
  });

  const client = new Client(
    {
      name: 'agent-platform',
      version: '0.1.0',
    },
    { capabilities: {} },
  );

  await client.connect(transport);

  // 列出所有可用工具
  const { tools } = await client.listTools();
  return { client, tools };
}

// 调用 MCP 工具
export async function callOrderQuery(
  client: Client,
  input: {
    order_id?: string;
    user_id?: string;
  },
) {
  const result = await client.callTool({
    name: 'order_query',
    arguments: input,
  });
  return result;
}
```

### 工具注册到 Agent

```typescript
// src/tools/registry.ts
import { orderQueryTool } from './builtin/order-query';
import { refundApplyTool } from './builtin/refund-apply';
import { connectOrderSystem } from './mcp/connect-order-system';

export async function buildToolRegistry() {
  // 1. 注册 builtin 工具
  const builtinTools = [orderQueryTool, refundApplyTool];

  // 2. 连 MCP 服务，把 MCP 工具也并进来
  const { client, tools: mcpTools } = await connectOrderSystem();
  const allTools = [...builtinTools, ...mcpTools];

  // 3. 返回给 Agent 循环用
  return {
    tools: allTools,
    cleanup: () => client.close(),
  };
}
```
