## 13.5 Koa 路由 + Vercel AI SDK（串起来）

```typescript
// src/gateway/chat-routes.ts
import Router from '@koa/router';
import { z } from 'zod';
import { streamText, convertToCoreMessages, type UIMessage } from 'ai';
import { anthropic } from '@ai-sdk/anthropic';
import { Readable } from 'node:stream';

const chatInputSchema = z.object({
  tenant_id: z.string().min(1),
  session_id: z.string().uuid(),
  messages: z.array(
    z.object({
      role: z.enum(['user', 'assistant']),
      content: z.string(),
    }),
  ),
});

export function buildChatRouter() {
  const router = new Router();

  // 同步版（给非浏览器调用方用）
  router.post('/api/v1/chat', async (ctx) => {
    const { tenant_id, session_id, messages } = chatInputSchema.parse(ctx.request.body);

    const result = await streamText({
      model: anthropic('claude-sonnet-4-5'),
      system: await loadSystemPrompt(tenant_id),
      messages: convertToCoreMessages(messages as UIMessage[]),
    });

    const text = await result.text; // 完整文本（流式版本用 toDataStreamResponse）
    ctx.body = { reply: text, session_id };
  });

  // 流式版（SSE，给浏览器用）
  router.post('/api/v1/chat/stream', async (ctx) => {
    const { tenant_id, messages } = chatInputSchema.parse(ctx.request.body);

    const result = streamText({
      model: anthropic('claude-sonnet-4-5'),
      system: await loadSystemPrompt(tenant_id),
      messages: convertToCoreMessages(messages as UIMessage[]),
    });

    const response = result.toDataStreamResponse();
    ctx.status = response.status;
    response.headers.forEach((v, k) => ctx.set(k, v));
    ctx.body = response.body ? Readable.fromWeb(response.body) : null;
  });

  return router;
}

// src/server.ts
import Koa from 'koa';
import bodyParser from 'koa-bodyparser';
import cors from '@koa/cors';
import { runMigrations } from '@/db';
import { buildChatRouter } from '@/gateway/chat-routes';

async function bootstrap() {
  // 1. 跑迁移
  await runMigrations();

  // 2. 启 Koa（Vercel AI SDK 的 provider 在路由内按需拿）
  const app = new Koa();
  app.use(cors());
  app.use(bodyParser());
  app.use(buildChatRouter().routes());

  app.listen(3000);
  console.log('Server running on http://localhost:3000');
}

bootstrap().catch(console.error);
```

## 13.6 SSE 流式聊天（Koa + Vercel AI SDK）

> 聊天界面要「边生成边显示」，用 SSE。后端用 Vercel AI SDK 的 `streamText()` 一句拿流式，前端用 `@ai-sdk/react` 的 `useChat()` 一行接。
> 比直接调 Anthropic SDK 简单 ~60 行。

```typescript
// src/gateway/chat-stream-routes.ts
import Router from '@koa/router';
import { z } from 'zod';
import { streamText, convertToCoreMessages, type UIMessage } from 'ai';
import { anthropic } from '@ai-sdk/anthropic';
import { Readable } from 'node:stream';

const chatInputSchema = z.object({
  tenant_id: z.string().min(1), // 租户号
  session_id: z.string().uuid(),
  messages: z.array(
    z.object({
      role: z.enum(['user', 'assistant']),
      content: z.string(),
    }),
  ), // 来自前端 useChat 的 messages 数组
});

export function buildChatStreamRouter() {
  const router = new Router();

  router.post('/api/v1/chat/stream', async (ctx) => {
    const { tenant_id, session_id, messages } = chatInputSchema.parse(ctx.request.body);

    // Vercel AI SDK 一句调用
    const result = streamText({
      model: anthropic('claude-sonnet-4-5'),
      system: await loadSystemPrompt(tenant_id),
      messages: convertToCoreMessages(messages as UIMessage[]),
      // 可选：工具调用
      // tools: { order_query: tool({...}) },
      // maxTokens: 1024,
      // temperature: 0.2,
    });

    // Vercel AI SDK 返回标准 Response 对象，转成 Koa 流
    const response = result.toDataStreamResponse();
    ctx.status = response.status;
    response.headers.forEach((v, k) => ctx.set(k, v));
    ctx.body = response.body ? Readable.fromWeb(response.body) : null;
  });

  return router;
}
```

> **对比之前**（手写 Anthropic SDK + SSE 响应头 + event 循环）：从 30+ 行降到 5 行。
> **之前手写的循环**（for await + event 分类 + 手动写 `event: text`）**全部不需要**——Vercel AI SDK 自动生成正确的 SSE 格式。

### 加上工具调用（Vercel AI SDK 的 `tool()` helper）

```typescript
// src/agent-orchestrator/tools.ts
import { tool } from 'ai';
import { z } from 'zod';

export const orderQueryTool = tool({
  description: '查询用户的订单',
  parameters: z.object({
    order_id: z.string().optional(),
    user_id: z.string().optional(),
  }),
  execute: async ({ order_id, user_id }) => {
    return await mcpClient.callTool('order_query', { order_id, user_id });
  },
});

// 在 streamText 里挂上
const result = streamText({
  model: anthropic('claude-sonnet-4-5'),
  system: systemPrompt,
  messages: convertToCoreMessages(messages),
  tools: { order_query: orderQueryTool },
  // Vercel AI SDK 自动循环：模型说要工具 → 调 execute → 把结果喂回去 → 继续生成
});
```

> **对比之前**（手写 tool call 循环）：Vercel AI SDK **自动**处理工具调用的全循环，**工具结果自动回灌**给模型。
